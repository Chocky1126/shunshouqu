```python
from pathlib import Path
import shutil, zipfile

archive = Path("outputs/顺手取-v1.6.7-安卓源码.zip")
target = Path("work/pickup-app-v1.6.8")
if target.exists():
    raise SystemExit(f"refusing to overwrite {target}")
with zipfile.ZipFile(archive) as z:
    names = [n for n in z.namelist() if not n.endswith("/")]
    prefix = names[0].split("/", 1)[0] + "/"
    for name in names:
        relative = name[len(prefix):]
        destination = target / relative
        destination.parent.mkdir(parents=True, exist_ok=True)
        destination.write_bytes(z.read(name))
```

核心实现必须等价于：

```java
final class Onboarding {
  static final String PREFS = "onboarding";
  static final int VERSION = 2;
  private static final String SEEN_VERSION = "seen_version";

  static boolean shouldShow(Context context) {
    return context.getSharedPreferences(PREFS, Context.MODE_PRIVATE)
        .getInt(SEEN_VERSION, 0) < VERSION;
  }

  static void complete(Context context) {
    context.getSharedPreferences(PREFS, Context.MODE_PRIVATE)
        .edit().putInt(SEEN_VERSION, VERSION).apply();
  }
}
```
